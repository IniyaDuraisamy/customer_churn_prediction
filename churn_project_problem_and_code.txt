Customer Churn Prediction Project
====================================================================================

1. Project Problem Statement
----------------------------
Organizations that provide subscription or recurring services (telecom, SaaS, streaming)
face customer churn — users who stop subscribing. Predicting churn early enables targeted
retention actions. The goal of this project is to build a reliable churn prediction model
and provide transparent, actionable explanations for predictions using SHAP.

Key requirements:
- Input: a customer-level dataset with a binary churn target (0 = retained, 1 = churned).
- Output: trained model, evaluation metrics (AUC, F1, calibration), global SHAP analysis,
  and local SHAP explanations for representative customers.

2. Project Goals
----------------
Primary goals:
1. Engineer meaningful features that capture customer behavior relevant to churn.
2. Train and validate a high-performing classification model (LightGBM or XGBoost).
3. Produce global interpretability using SHAP summary and dependence plots.
4. Produce local interpretability using SHAP force and waterfall plots for five customers.
5. Package deliverables: model file, plots, CSVs of predictions and SHAP contributions, and a short textual report.

```python
# TASK 1: Feature Engineering
import pandas as pd
import numpy as np
import os

DATA_PATH = "customer_data.csv"  # change if needed
if os.path.exists(DATA_PATH):
    df = pd.read_csv(DATA_PATH)
else:
    raise FileNotFoundError(f"{DATA_PATH} not found. Provide your dataset or run a mock data generator.")

# Ensure numeric conversions
for c in ["TotalCharges","MonthlyCharges","tenure"]:
    if c in df.columns:
        df[c] = pd.to_numeric(df[c], errors='coerce')

# Fill numeric missing with median
num_cols = df.select_dtypes(include=[np.number]).columns.tolist()
for c in num_cols:
    df[c].fillna(df[c].median(), inplace=True)

# Fill categorical missing
cat_cols = df.select_dtypes(include=['object','category']).columns.tolist()
for c in cat_cols:
    df[c].fillna('missing', inplace=True)

# Feature 1: recency_days (if last_transaction_date exists)
if 'last_transaction_date' in df.columns:
    df['last_transaction_date'] = pd.to_datetime(df['last_transaction_date'], errors='coerce')
    snapshot = df['last_transaction_date'].max()
    df['recency_days'] = (snapshot - df['last_transaction_date']).dt.days.fillna(999)
else:
    df['recency_days'] = 999

# Feature 2: tenure_days (use existing tenure if present)
if 'tenure' in df.columns:
    df['tenure_days'] = df['tenure']
else:
    df['tenure_days'] = 0

# Feature 3: avg_charges_per_tenure
df['avg_charges_per_tenure'] = df.apply(
    lambda r: r['TotalCharges'] / (r['tenure'] if ('tenure' in r and r['tenure']>0) else 1), axis=1
)

# Feature 4: calls_per_100days (if num_support_calls exists)
if 'num_support_calls' in df.columns:
    df['calls_per_100days'] = df['num_support_calls'] / (df['tenure'].replace(0,1) / 100 + 1e-9)
else:
    df['calls_per_100days'] = 0

# Feature 5: usage_change_ratio (if prev/current usage exist)
if 'prev_month_usage' in df.columns and 'current_month_usage' in df.columns:
    df['usage_change_ratio'] = (df['current_month_usage'] - df['prev_month_usage']) / (df['prev_month_usage'] + 1e-9)
else:
    df['usage_change_ratio'] = 0

# Additional useful features
if 'MonthlyCharges' in df.columns:
    df['high_monthly_charge'] = (df['MonthlyCharges'] > df['MonthlyCharges'].median()).astype(int)
if 'PaymentMethod' in df.columns:
    df['payment_electronic_check'] = (df['PaymentMethod'] == 'Electronic check').astype(int)

# Save processed snapshot
OUT_DIR = "outputs"
os.makedirs(OUT_DIR, exist_ok=True)
df.to_csv(os.path.join(OUT_DIR, "data_processed_task1.csv"), index=False)
print("TASK 1 complete. Processed data saved to outputs/data_processed_task1.csv")
```

# Deliverable 1: End-to-end pipeline
import os
import numpy as np
import pandas as pd
import joblib
import matplotlib.pyplot as plt

from sklearn.model_selection import train_test_split
from sklearn.preprocessing import LabelEncoder
from sklearn.metrics import roc_auc_score, f1_score, classification_report, confusion_matrix, brier_score_loss

import lightgbm as lgb
import shap

OUTPUT_DIR = "outputs"
os.makedirs(OUTPUT_DIR, exist_ok=True)
RANDOM_STATE = 42
np.random.seed(RANDOM_STATE)

# 1) Load dataset or generate mock data if missing
DATA_PATH = "customer_data.csv"
if os.path.exists(DATA_PATH):
    df = pd.read_csv(DATA_PATH)
else:
    # Mock Telco-like dataset generator
    def make_mock_telco(n=5000, seed=RANDOM_STATE):
        np.random.seed(seed)
        dfm = pd.DataFrame({
            "customerID": [f"C{100000+i}" for i in range(n)],
            "tenure": np.random.randint(0,72,size=n),
            "MonthlyCharges": np.round(np.random.uniform(18,120,size=n),2),
            "TotalCharges": 0.0,
            "TechSupport": np.random.choice(["Yes","No","No internet service"], size=n, p=[0.2,0.65,0.15]),
            "PaymentMethod": np.random.choice(["Electronic check","Mailed check","Bank transfer (automatic)","Credit card (automatic)"], size=n)
        })
        dfm["TotalCharges"] = np.round(dfm["MonthlyCharges"] * dfm["tenure"] + np.random.normal(0,20,size=n),2).clip(0)
        prob = (
            0.25*(dfm["tenure"] < 6).astype(float) +
            0.20*(dfm["PaymentMethod"]=="Electronic check").astype(float) +
            0.15*(dfm["MonthlyCharges"] > dfm["MonthlyCharges"].median()).astype(float) +
            0.12*(dfm["TechSupport"] == "No").astype(float)
        )
        prob = np.clip(prob + np.random.normal(0,0.06,size=n), 0, 1)
        dfm["churn"] = (np.random.rand(n) < prob).astype(int)
        return dfm
    df = make_mock_telco(n=5000)

# 2) Basic cleaning
for c in ["TotalCharges","MonthlyCharges","tenure"]:
    if c in df.columns:
        df[c] = pd.to_numeric(df[c], errors='coerce')
num_cols = df.select_dtypes(include=[np.number]).columns.tolist()
for c in num_cols:
    df[c].fillna(df[c].median(), inplace=True)
cat_cols = df.select_dtypes(include=['object','category']).columns.tolist()
for c in cat_cols:
    df[c].fillna('missing', inplace=True)

# 3) Feature engineering (>=5 meaningful features)
df['avg_charges_per_tenure'] = df.apply(lambda r: r['TotalCharges'] / (r['tenure'] if r.get('tenure',0) > 0 else 1), axis=1)
df['high_monthly_charge'] = (df['MonthlyCharges'] > df['MonthlyCharges'].median()).astype(int)
df['has_tech_support'] = (df['TechSupport'] == 'Yes').astype(int) if 'TechSupport' in df.columns else 0
df['payment_electronic_check'] = (df['PaymentMethod'] == 'Electronic check').astype(int) if 'PaymentMethod' in df.columns else 0
df['tenure_short'] = (df['tenure'] < 6).astype(int)
stream_cols = [c for c in ['StreamingTV','StreamingMovies'] if c in df.columns]
if stream_cols:
    df['streaming_count'] = df[stream_cols].apply(lambda row: sum([1 if v=='Yes' else 0 for v in row]), axis=1)
else:
    df['streaming_count'] = 0

# 4) Prepare modeling dataset
id_cols = [c for c in df.columns if c.lower() in ('customerid','customer_id','id')]
df_model = df.drop(columns=id_cols, errors='ignore')
if 'churn' not in df_model.columns:
    raise RuntimeError("Target 'churn' missing.")
X = df_model.drop(columns=['churn'])
y = df_model['churn'].astype(int)

# 5) Label encode categorical columns
cat_cols = X.select_dtypes(include=['object','category']).columns.tolist()
le_map = {}
for c in cat_cols:
    le = LabelEncoder()
    X[c] = le.fit_transform(X[c].astype(str))
    le_map[c] = le

# 6) Train/test split (stratified)
X_train, X_test, y_train, y_test = train_test_split(X, y, test_size=0.2, stratify=y, random_state=RANDOM_STATE)

# Save test set for deliverables
X_test.to_csv(os.path.join(OUTPUT_DIR, "X_test.csv"), index=False)
y_test.reset_index(drop=True).to_frame().to_csv(os.path.join(OUTPUT_DIR, "y_test.csv"), index=False)

# 7) Train LightGBM
lgb_train = lgb.Dataset(X_train, label=y_train, categorical_feature=[c for c in cat_cols], free_raw_data=False)
lgb_val = lgb.Dataset(X_test, label=y_test, reference=lgb_train, free_raw_data=False)
params = {'objective':'binary','metric':'auc','boosting_type':'gbdt','learning_rate':0.05,'num_leaves':31,'feature_fraction':0.8,'bagging_fraction':0.8,'bagging_freq':5,'seed':RANDOM_STATE,'verbosity':-1}
model = lgb.train(params, lgb_train, num_boost_round=1000, valid_sets=[lgb_train, lgb_val], early_stopping_rounds=50, verbose_eval=100)
joblib.dump(model, os.path.join(OUTPUT_DIR, "final_model.joblib"))

# 8) Evaluate and save metrics & predictions
y_proba = model.predict(X_test, num_iteration=model.best_iteration)
y_pred = (y_proba >= 0.5).astype(int)
auc = roc_auc_score(y_test, y_proba)
f1 = f1_score(y_test, y_pred)
brier = brier_score_loss(y_test, y_proba)
pd.DataFrame({'metric':['auc','f1','brier'],'value':[auc,f1,brier]}).to_csv(os.path.join(OUTPUT_DIR,"evaluation_metrics.csv"), index=False)
pd.DataFrame({'y_true': y_test.reset_index(drop=True), 'y_proba': y_proba, 'y_pred': y_pred}).to_csv(os.path.join(OUTPUT_DIR,"test_predictions.csv"), index=False)

# 9) Global SHAP analysis
explainer = shap.TreeExplainer(model, data=X_train, model_output="probability")
shap_values = explainer.shap_values(X_test)
shap.summary_plot(shap_values, X_test, plot_type="dot", max_display=20, show=False)
plt.tight_layout(); plt.savefig(os.path.join(OUTPUT_DIR, "shap_summary_dot.png")); plt.close()
shap.summary_plot(shap_values, X_test, plot_type="bar", max_display=20, show=False)
plt.tight_layout(); plt.savefig(os.path.join(OUTPUT_DIR, "shap_summary_bar.png")); plt.close()
import numpy as np
mean_abs = np.abs(shap_values).mean(axis=0)
feat_rank = pd.DataFrame({'feature': X_test.columns, 'mean_abs_shap': mean_abs}).sort_values('mean_abs_shap', ascending=False)
feat_rank.to_csv(os.path.join(OUTPUT_DIR, "shap_feature_ranking.csv"), index=False)

# 10) Local SHAP explanations (5 customers)
test_df = X_test.reset_index(drop=True).copy()
test_df['pred_proba'] = y_proba
test_df['true'] = y_test.reset_index(drop=True)
churned_sel = test_df[test_df['true']==1].sort_values('pred_proba', ascending=False).head(3).index.tolist()
retained_sel = test_df[test_df['true']==0].sort_values('pred_proba', ascending=True).head(2).index.tolist()
selected_indices = churned_sel + retained_sel
base_value = explainer.expected_value if not isinstance(explainer.expected_value, list) else explainer.expected_value[1]
local_rows = []
shap.initjs()
for idx in selected_indices:
    row = X_test.reset_index(drop=True).iloc[idx:idx+1]
    row_shap = shap_values[idx]
    force_html = shap.force_plot(base_value, row_shap, row, matplotlib=False)
    shap.save_html(os.path.join(OUTPUT_DIR, f"force_plot_{idx}.html"), force_html)
    plt.figure()
    try:
        shap.plots._waterfall.waterfall_legacy(base_value, row_shap, row.iloc[0], max_display=12)
    except Exception:
        contribs = pd.DataFrame({'feature': X_test.columns, 'shap_value': row_shap, 'abs_shap': np.abs(row_shap), 'feature_value': row.iloc[0].values})
        contribs = contribs.sort_values('abs_shap', ascending=False).head(12)
        contribs.plot.barh(x='feature', y='shap_value')
    plt.tight_layout(); plt.savefig(os.path.join(OUTPUT_DIR, f"waterfall_{idx}.png")); plt.close()
    feat_df = pd.DataFrame({'feature': X_test.columns, 'shap_value': row_shap, 'abs_shap': np.abs(row_shap), 'feature_value': row.iloc[0].values}).sort_values('abs_shap', ascending=False).head(10)
    feat_df['test_row'] = idx
    local_rows.append(feat_df)
pd.concat(local_rows, ignore_index=True).to_csv(os.path.join(OUTPUT_DIR, "selected_customers_top_shap.csv"), index=False)

print("Finished. Check the 'outputs' folder for model, metrics, SHAP plots, and local explanations.")
